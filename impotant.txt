@frappe.whitelist(allow_guest=True)  # allow_guest=True rakhvathi login vagar na users pan form fill kari shakse
def save_custom_form(name, email, message):
    try:
        # Navo document/record create karva mate
        new_doc = frappe.get_doc({
            "doctype": "Customer Query",  # Step 1 ma banavelu DocType nu name exact lakhvu
            "full_name": name,
            "email": email,
            "message": message
        })
        
        # Database ma data insert (save) karva mate
        new_doc.insert(ignore_permissions=True)
        
        return {"status": "success", "message": "Data successfully store thai gayo!"}
        
    except Exception as e:
        return {"status": "error", "message": str(e)}
